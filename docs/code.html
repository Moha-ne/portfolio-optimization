import numpy as np
import pandas as pd
import yfinance as yf
import plotly.graph_objects as go

images = Path(par84.img_path)

df = yf.Tickers(symbols).history(period='3y', auto_adjust=True)['Close']

df100 = df / df.iloc[0] * 100
print(df100)

fig = go.Figure(layout=par84.layout)
for i in symbols:

print(f"\n==== Daily returns ====")
daily_returns = np.log(df/ df.shift(1)).dropna()
print(daily_returns.map(lambda x: f"{x:.2%}"))

fig = go.Figure(layout=par84.layout)
for i in symbols:
    fig.add_scatter(x=daily_returns.index, y=daily_returns[i], name=i)

fig = go.Figure(layout=par84.layout)	
for i in symbols:
    fig.add_histogram(x=daily_returns[i], name=i)
fig.write_html(images / f"{name}3.html", config=par84.config)

print(f"\n==== Annual returns ====")
mean_returns = daily_returns.mean() * 252
print(mean_returns.map(lambda x: f"{x:.2%}"))

print(f"\n==== Covariance matrix ====")
cov_matrix = daily_returns.cov() * 252
print(cov_matrix.map(lambda x: f"{x:.2%}"))

print(f"\n==== Risk-free rate ====")

rf = rf_series.dropna().mean() / 100.0

np.random.seed(42)

results = np.zeros((num_portfolios, 3))

    weights = np.random.random(len(symbols))
    weights /= np.sum(weights)

2.py (GitHub)

Momentum, volatility, beta and composite z-score of each asset.

import numpy as np
import pandas as pd
import yfinance as yf

tickers = ["SHY", "GLD", "KO", "CBOE", "MO", "DBA", "DBMF", "TIP", "VCSH", "PM", "TRV", "MUB", "MNA", "LLY", "ABBV", "IBM", "WMT", "IEF", "JNJ", "V", "AEP", "ALL", "AAPL", "NDAQ", "MSFT"]

benchmark = "SPY"

data = yf.Tickers(tickers + [benchmark]).history(period="3y", auto_adjust=True)['Close']

missing = [t for t in tickers + [benchmark] if t not in data.columns]
if missing:
    raise ValueError(f"Missing data for: {missing}")

prices = data[tickers]
market = data[benchmark]

returns = prices.pct_change().dropna()
market_returns = market.pct_change().dropna()

df = pd.DataFrame(index=tickers)

df["momentum"] = prices.shift(21).iloc[-1] / prices.shift(273).iloc[-1] - 1
df["volatility"] = returns.tail(252).std() * np.sqrt(252)

betas = []

for ticker in tickers:
    concat_returns = pd.concat([returns[ticker], market_returns], axis=1).dropna().tail(252)
    stock_ret = concat_returns.iloc[ : , 0]
    bench_ret = concat_returns.iloc[ : , 1]

    cov = np.cov(stock_ret, bench_ret, ddof=1)[0, 1]
    var_bench = np.var(bench_ret, ddof=1)
    beta = cov / var_bench

    betas.append(beta)

df["beta"] = betas
df = df.fillna(df.median(numeric_only=True))

def zscore(series):
    return (series - series.mean()) / series.std(ddof=0)

df["z_momentum"] = zscore(df["momentum"])
df["z_volatility"] = zscore(df["volatility"])
df["z_beta"] = zscore(df["beta"])

df["z_score"] = (
    0.5 * df["z_momentum"]
    - 0.3 * df["z_volatility"]
    - 0.2 * df["z_beta"]
)

df = df.sort_values("z_score", ascending=False)

print(df[["momentum", "volatility", "beta", "z_momentum", "z_volatility", "z_beta", "z_score"]])

3.py (GitHub)

Beta and alpha against the S&P 500.

import yfinance as yf
import numpy as np

tickers = ["SHY", "GLD", "KO", "CBOE", "MO", "DBA", "DBMF", "TIP", "VCSH", "PM", "TRV", "MUB", "MNA", "LLY", "ABBV", "IBM", "WMT", "IEF", "JNJ", "V", "AEP", "ALL", "AAPL", "NDAQ", "MSFT"]
benchmark = '^GSPC'

rf_annual = yf.Ticker('^IRX').history(period='5d')['Close'].dropna().iloc[-1] / 100
rf_daily = (1 + rf_annual)**(1/252) - 1

df = yf.Tickers(tickers + [benchmark]).history(period='3y', auto_adjust=True)['Close']

daily_returns = df.pct_change().dropna()
portfolio_returns = daily_returns[tickers].mean(axis=1)   # equal-weight portfolio
market_returns = daily_returns[benchmark]

beta = portfolio_returns.cov(market_returns) / market_returns.var()
alpha_daily = (portfolio_returns.mean() - rf_daily) - beta * (market_returns.mean() - rf_daily)
alpha_annual = (1 + alpha_daily)**252 - 1

print(f"Portfolio Beta: {beta:.4f}")
print(f"Daily Alpha: {alpha_daily:.6%}")
print(f"Annualized Alpha: {alpha_annual:.2%}")

from pathlib import Path
import yfinance as yf
import par84

images = Path(par84.img_path)
name = "news-"

tickers = ["CBOE", "KO", "MO"]
n = 3
html = ""

for ticker in tickers:
    news = yf.Search(ticker, news_count=n).news
    links = ""

    for x in news[:n]:
        link = x.get("link", "")
        title = x.get("title", "")
        links += f'<li><a href="{link}" target="_blank" style="color:white">{title}</a></li>'

    html += f"<h2>{ticker}</h2><ul>{links}</ul>"

    return (series - series.mean()) / series.std(ddof=0)

df["z_momentum"] = zscore(df["momentum"])
df["z_volatility"] = zscore(df["volatility"])
df["z_beta"] = zscore(df["beta"])

df["z_score"] = (
    0.6 * df["z_momentum"]
    - 0.2 * df["z_volatility"]
    - 0.2 * df["z_beta"]
)

df = df.sort_values("z_score", ascending=False)

print(df[["momentum", "volatility", "beta", "z_momentum", "z_volatility", "z_beta", "z_score"]])

Beta and alpha against the S&P 500.

import yfinance as yf
import numpy as np

tickers = ["PGR", "LLY", "JPM", "MU", "BRK-B", "RTX", "PLTR", "CVNA", "GE", "TMUS", "WMT", "CRWD", "GOOGL", "TSM", "IBKR", "AAPL", "GS", "PWR", "SPOT", "NVDA", "COST", "RCL", "GLD", "GDX", "NEM"]
benchmark = '^GSPC'

rf_annual = yf.Ticker('^IRX').history(period='5d')['Close'].dropna().iloc[-1] / 100
rf_daily = (1 + rf_annual)**(1/252) - 1

df = yf.Tickers(tickers + [benchmark]).history(period='3y', auto_adjust=True)['Close']

daily_returns = df.pct_change().dropna()
portfolio_returns = daily_returns[tickers].mean(axis=1)   # equal-weight portfolio
market_returns = daily_returns[benchmark]

beta = portfolio_returns.cov(market_returns) / market_returns.var()
alpha_daily = (portfolio_returns.mean() - rf_daily) - beta * (market_returns.mean() - rf_daily)
alpha_annual = (1 + alpha_daily)**252 - 1

print(f"Portfolio Beta: {beta:.4f}")
print(f"Daily Alpha: {alpha_daily:.6%}")
print(f"Annualized Alpha: {alpha_annual:.2%}")
